
import numpy as np
import tensorflow as tf
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import LSTM, Dense, Dropout
from tensorflow.keras.callbacks import ModelCheckpoint
import string
import requests


# 1. Dataset Loading & Preprocessing


url = "https://raw.githubusercontent.com/karpathy/char-rnn/master/data/tinyshakespeare/input.txt"
text = requests.get(url).text.lower()

# Reduce data size to avoid memory issues (important)
text = text[:300000]

# Remove punctuation (character-based cleaning)
text = text.translate(str.maketrans('', '', string.punctuation))

# Create character mappings
chars = sorted(list(set(text)))
char_to_int = {c: i for i, c in enumerate(chars)}
int_to_char = {i: c for i, c in enumerate(chars)}

n_vocab = len(chars)
print("Total Characters:", len(text))
print("Total Unique Characters:", n_vocab)


# 2. Create Training Sequences


seq_length = 100
dataX = []
dataY = []

for i in range(len(text) - seq_length):
    seq_in = text[i:i + seq_length]
    seq_out = text[i + seq_length]
    dataX.append([char_to_int[c] for c in seq_in])
    dataY.append(char_to_int[seq_out])

X = np.reshape(dataX, (len(dataX), seq_length, 1))
X = X / float(n_vocab)

y = tf.keras.utils.to_categorical(dataY, num_classes=n_vocab)

print("Total Patterns:", X.shape[0])


# 3. Model Definition


model = Sequential([
    LSTM(256, input_shape=(seq_length, 1), return_sequences=True),
    Dropout(0.2),
    LSTM(256),
    Dropout(0.2),
    Dense(n_vocab, activation="softmax")
])

model.compile(loss="categorical_crossentropy", optimizer="adam")
model.summary()


# 4. Model Training


checkpoint = ModelCheckpoint(
    "best_model.h5",
    monitor="loss",
    verbose=1,
    save_best_only=True,
    mode="min"
)

model.fit(X, y, epochs=10, batch_size=128, callbacks=[checkpoint])


# 5. Temperature Sampling Function


def sample(preds, temperature=0.8):
    preds = np.asarray(preds).astype("float64")
    preds = np.log(preds + 1e-8) / temperature
    exp_preds = np.exp(preds)
    preds = exp_preds / np.sum(exp_preds)
    return np.random.choice(len(preds), p=preds)


# 6. Text Generation Function


def generate_text(model, seed_text, length=300):
    seed_text = seed_text.lower()
    pattern = [char_to_int[c] for c in seed_text if c in char_to_int]

    # Ensure correct length
    pattern = pattern[-seq_length:]

    generated = seed_text

    for _ in range(length):
        x = np.reshape(pattern, (1, seq_length, 1))
        x = x / float(n_vocab)

        prediction = model.predict(x, verbose=0)[0]
        index = sample(prediction, temperature=0.7)
        result = int_to_char[index]

        generated += result
        pattern.append(index)
        pattern = pattern[1:]

    return generated


# 7. Generate Text


seed = "shall i compare the to a summers day "
print("\n--- GENERATED TEXT ---\n")
print(generate_text(model, seed))
