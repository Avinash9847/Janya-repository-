
import numpy as np
import tensorflow as tf
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import LSTM, Dense, Embedding, Dropout
from tensorflow.keras.callbacks import ModelCheckpoint
import string
import requests


# 1. Dataset Loading and Preprocessing

# Use Shakespeare's
url = "https://raw.githubusercontent.com/karpathy/char-rnn/master/data/tinyshakespeare/input.txt"
text = requests.get(url).text.lower()

# Remove punctuation for cleaner word-based generation (optional for character-based)
text = text.translate(str.maketrans('', '', string.punctuation))

# Create character mappings
chars = sorted(list(set(text)))
char_to_int = {c: i for i, c in enumerate(chars)}
int_to_char = {i: c for i, c in enumerate(chars)}

n_chars = len(text)
n_vocab = len(chars)
print(f"Total Characters: {n_chars}")
print(f"Total Vocab (Unique Characters): {n_vocab}")


seq_length = 100
dataX = []
dataY = []


for i in range(0, n_chars - seq_length, 1):
    seq_in = text[i:i + seq_length]
    seq_out = text[i + seq_length]
    dataX.append([char_to_int[char] for char in seq_in])
    dataY.append(char_to_int[seq_out])

n_patterns = len(dataX)
print(f"Total Patterns: {n_patterns}")

# Reshape X to be [samples, time steps, features]
X = np.reshape(dataX, (n_patterns, seq_length, 1))
# Normalize input
X = X / float(n_vocab)
# One-hot encode the output variable
y = tf.keras.utils.to_categorical(dataY)


# 2. Model Design

model = Sequential([
    
    LSTM(256, input_shape=(X.shape[1], X.shape[2]), return_sequences=True),
    Dropout(0.2),
    LSTM(256),
    Dropout(0.2),
    
    Dense(y.shape[1], activation='softmax')
])

model.compile(loss='categorical_crossentropy', optimizer='adam')

# 3. Model Training

filepath = "weights-improvement.hdf5"
checkpoint = ModelCheckpoint(filepath, monitor='loss', verbose=1, save_best_only=True, mode='min')
callbacks_list = [checkpoint]

model.fit(X, y, epochs=10, batch_size=128, callbacks=callbacks_list)


# 4. Text Generation Logic


def generate_text(model, seed_text, n_to_generate=200):
    pattern = [char_to_int[char] for char in seed_text.lower() if char in char_to_int]
    
  
    if len(pattern) < seq_length:
        pattern = [0] * (seq_length - len(pattern)) + pattern
    pattern = pattern[-seq_length:]
    
    generated_output = ""
    
    for i in range(n_to_generate):
        x = np.reshape(pattern, (1, len(pattern), 1))
        x = x / float(n_vocab)
        
        prediction = model.predict(x, verbose=0)
        index = np.argmax(prediction)
        result = int_to_char[index]
        
        generated_output += result
        pattern.append(index)
        pattern = pattern[1:len(pattern)]
        
    return generated_output


print("\n--- GENERATED TEXT ---")
seed = "shall i compare thee to a summers day "
print(f"Seed: {seed}")
print(generate_text(model, seed))
